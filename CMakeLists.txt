cmake_minimum_required(VERSION 3.20)
project(AAAnimalStudio VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(aaanimal_core STATIC
  native/platform.c native/image.c native/mesh.c native/simplify.c
  native/style.c native/body.c native/shape.c native/detail.c native/observed.c
  native/volume.c native/form.c native/export.c native/pipeline.c native/references.c native/gbif.c
  vendor/cjson/cJSON.c)
target_include_directories(aaanimal_core PUBLIC native)
target_compile_options(aaanimal_core PRIVATE -Wall -Wextra -Wno-misleading-indentation -Wno-deprecated-declarations)
target_link_libraries(aaanimal_core PUBLIC CURL::libcurl Threads::Threads m)
if(APPLE)
  target_link_libraries(aaanimal_core PUBLIC "-framework CoreFoundation" "-framework CoreGraphics" "-framework ImageIO")
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(aaanimal_core PUBLIC OpenSSL::Crypto)
endif()
add_executable(aaanimal-cli native/cli.c)
target_link_libraries(aaanimal-cli PRIVATE aaanimal_core)
enable_testing()
add_executable(native-tests native/tests.c)
target_compile_options(native-tests PRIVATE -UNDEBUG)
target_link_libraries(native-tests PRIVATE aaanimal_core)
add_test(NAME core COMMAND native-tests)
add_executable(platform-tests native/platform_tests.c)
target_compile_options(platform-tests PRIVATE -UNDEBUG)
target_link_libraries(platform-tests PRIVATE aaanimal_core)
add_test(NAME platform COMMAND platform-tests)
set_tests_properties(core platform PROPERTIES TIMEOUT 180)
if(APPLE)
  enable_language(OBJC)
  add_executable(AAAnimalStudio MACOSX_BUNDLE macos/App.m)
  target_compile_options(AAAnimalStudio PRIVATE -fobjc-arc)
  target_link_libraries(AAAnimalStudio PRIVATE aaanimal_core "-framework Cocoa" "-framework SceneKit")
  set_target_properties(AAAnimalStudio PROPERTIES
    OUTPUT_NAME "AAAnimal Studio"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist")
endif()
